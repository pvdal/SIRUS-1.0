/**
 * Preparando o ambiente Ubuntu
 */

# PHP:
sudo add-apt-repository ppa:ondrej/php -y
sudo apt update
sudo apt install -y php8.3-fpm php8.3-cli php8.3-mysql php8.3-xml php8.3-curl php8.3-mbstring php8.3-zip php8.3-bcmath php8.3-intl php8.3-gd php8.3-readline php8.3-soap php8.3-imagick unzip

# COMPOSER:
sudo apt install -y curl unzip git
curl -sS https://getcomposer.org/installer -o composer-setup.php
sudo php composer-setup.php --install-dir=/usr/local/bin --filename=composer

# NODE:
curl -fsSL https://deb.nodesource.com/setup_22.x | sudo -E bash -
sudo apt install -y nodejs

# MYSQL:
sudo apt install -y mysql-server
sudo mysql_secure_installation
sudo mysql
- CREATE DATABASE sirus CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
- CREATE USER 'usuario'@'localhost' IDENTIFIED BY 'senha_forte';
- GRANT ALL PRIVILEGES ON sirus.* TO 'usuario'@'localhost';
- FLUSH PRIVILEGES;
- EXIT;

# NGINX:
sudo apt install nginx -y
sudo systemctl enable nginx
sudo systemctl start nginx

# SUPERVISOR:
sudo apt install supervisor -y
sudo systemctl enable supervisor
sudo systemctl start supervisor

# Verificações finais:
php -v
composer -V
node -v
mysql --version
nginx -v
